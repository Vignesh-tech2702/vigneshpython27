class Node:
    def __init__(self, data):
        self.data = data
        self.next = None


class Stack:
    def __init__(self):
        self.top = None

    def push(self, data):
        new_node = Node(data)
        new_node.next = self.top
        self.top = new_node
        print(data, "inserted into stack")

    def pop(self):
        if self.top is None:
            print("Stack Underflow")
        else:
            temp = self.top.data
            self.top = self.top.next
            print("Deleted item:", temp)

    def peek(self):
        if self.top is None:
            print("Stack is Empty")
        else:
            print("Top element:", self.top.data)

    def display(self):
        if self.top is None:
            print("Stack is Empty")
        else:
            print("Stack elements:")
            temp = self.top
            while temp:
                print(temp.data)
                temp = temp.next


s = Stack()

while True:
    print("\n1. Push")
    print("2. Pop")
    print("3. Peek")
    print("4. Display")
    print("5. Exit")

    choice = int(input("Enter your choice: "))

    if choice == 1:
        item = input("Enter book title: ")
        s.push(item)

    elif choice == 2:
        s.pop()

    elif choice == 3:
        s.peek()

    elif choice == 4:
        s.display()

    elif choice == 5:
        print("Program Ended")
        break

    else:
        print("Invalid Choice")
